-- Tạo cơ sở dữ liệu
CREATE DATABASE QLKYTHI;

GO

USE QLKYTHI;

-- Bảng MHOC (Môn học)
CREATE TABLE MHOC (
    MAMH VARCHAR(10) PRIMARY KEY,
    TENMH NVARCHAR(100),
    SOTIET INT
);

GO

-- Bảng GV (Giáo viên)
CREATE TABLE GV (
    MAGV VARCHAR(10) PRIMARY KEY,
    TENGV NVARCHAR(100),
    MAMH VARCHAR(10), -- MAMH có thể là null nếu không phải là chủ nhiệm
    FOREIGN KEY (MAMH) REFERENCES MHOC(MAMH)  -- Mối quan hệ với bảng MHOC
);

GO

-- Bảng BUOITHI (Buổi thi)
CREATE TABLE BUOITHI (
    HKY VARCHAR(10),
    NGAY DATE,
    GIO TIME,
    PHG VARCHAR(10),
    MAMH VARCHAR(10),
    TGTHI INT,
    PRIMARY KEY (HKY, NGAY, GIO, PHG),
    FOREIGN KEY (MAMH) REFERENCES MHOC(MAMH)  -- Mối quan hệ với bảng MHOC
);

GO

-- Bảng PC_COI_THI (Phân công coi thi)
CREATE TABLE PC_COI_THI (
    MAGV VARCHAR(10),
    HKY VARCHAR(10),
    NGAY DATE,
    GIO TIME,
    PHG VARCHAR(10),
    PRIMARY KEY (MAGV, HKY, NGAY, GIO, PHG),
    FOREIGN KEY (MAGV) REFERENCES GV(MAGV)  -- Mối quan hệ với bảng GV
);

GO

--a. Danh sách các giáo viên dạy các môn học có số tiết từ 45 trở lên
SELECT DISTINCT GV.TENGV
FROM GV JOIN MHOC ON GV.MAMH = MHOC.MAMH
WHERE MHOC.SOTIET >= 45;

--b. Danh sách giáo viên được phân công gác thi trong học kỳ 1
SELECT DISTINCT GV.TENGV
FROM GV JOIN PC_COI_THI ON GV.MAGV = PC_COI_THI.MAGV
WHERE PC_COI_THI.HKY = 'HK1';

--c. Danh sách giáo viên không được phân công gác thi trong học kỳ 1
SELECT TENGV
FROM GV
WHERE MAGV NOT IN (
    SELECT MAGV
    FROM PC_COI_THI
    WHERE HKY = 'HK1'
);

--d. Cho biết lịch thi môn văn (TENMH = ‘VĂN HỌC’)
SELECT *
FROM BUOITHI

--e. Cho biết các buổi gác thi của các giáo viên chủ nhiệm môn văn (TENMH = ‘VĂN HỌC’).
SELECT PC_COI_THI.*
FROM PC_COI_THI JOIN GV ON PC_COI_THI.MAGV = GV.MAGV
WHERE GV.MAMH = 'MH01';
