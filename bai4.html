-- Tạo CSDL
CREATE DATABASE GarageManagement;

GO

USE GarageManagement;

-- Bảng THO (Người thợ)
CREATE TABLE THO (
    MaTho VARCHAR(10) PRIMARY KEY,
    TenTho NVARCHAR(100),
    Nhom NVARCHAR(50),
    NhomTruong VARCHAR(10),
    FOREIGN KEY (NhomTruong) REFERENCES THO(MaTho)  -- Nhóm trưởng phải là một trong những người thợ
);

GO

-- Bảng CONGVIEC (Công việc)
CREATE TABLE CONGVIEC (
    MaCV VARCHAR(10) PRIMARY KEY,
    NoiDungCV NVARCHAR(255)
);

GO

-- Bảng KHACHHANG (Khách hàng)
CREATE TABLE KHACHHANG (
    MaKH VARCHAR(10) PRIMARY KEY,
    TenKH NVARCHAR(100),
    DiaChi NVARCHAR(255),
    DienThoai VARCHAR(15)
);

GO

-- Bảng HOPDONG (Hợp đồng)
CREATE TABLE HOPDONG (
    SoHD VARCHAR(10) PRIMARY KEY,
    NgayHD DATE,
    MaKH VARCHAR(10),
    SoXe VARCHAR(20),
    TriGiaHD DECIMAL(10, 2),
    NgayGiaoDK DATE,
    NgayNgThu DATE,
    FOREIGN KEY (MaKH) REFERENCES KHACHHANG(MaKH)
);

GO

-- Bảng CHITIET_HD (Chi tiết hợp đồng)
CREATE TABLE CHITIET_HD (
    SoHD VARCHAR(10),
    MaCV VARCHAR(10),
    TriGiaCV DECIMAL(10, 2),
    MaTho VARCHAR(10),
    KhoanTHo DECIMAL(10, 2),
    PRIMARY KEY (SoHD, MaCV),
    FOREIGN KEY (SoHD) REFERENCES HOPDONG(SoHD),
    FOREIGN KEY (MaCV) REFERENCES CONGVIEC(MaCV),
    FOREIGN KEY (MaTho) REFERENCES THO(MaTho)
);

GO

-- Bảng PHIEUTHU (Phiếu thu)
CREATE TABLE PHIEUTHU (
    SoPT VARCHAR(10) PRIMARY KEY,
    NgayLapPT DATE,
    SoHD VARCHAR(10),
    MaKH VARCHAR(10),
    HoTen NVARCHAR(100),
    SoTienThu DECIMAL(10, 2),
    FOREIGN KEY (SoHD) REFERENCES HOPDONG(SoHD),
    FOREIGN KEY (MaKH) REFERENCES KHACHHANG(MaKH)
);
GO

--1. Cho biết danh sách các người thợ hiện không tham gia vào một hợp đồng sửa chữa nào.
SELECT * 
FROM THO 
WHERE MaTho NOT IN (SELECT DISTINCT MaTho FROM CHITIET_HD);

--2. Cho biết danh sách những hợp đồng đã thanh lý nhưng chưa được thanh toán tiền đầy đủ.
SELECT H.*
FROM HOPDONG H
LEFT JOIN PHIEUTHU P ON H.SoHD = P.SoHD
WHERE H.NgayNgThu IS NOT NULL AND 
      (SELECT SUM(SoTienThu) FROM PHIEUTHU WHERE SoHD = H.SoHD) < H.TriGiaHD;

--3. Cho biết danh sách những hợp đồng cần phải hoàn tất trước ngày 31/12/2002.
SELECT *
FROM HOPDONG
WHERE NgayGiaoDK < '2023-3-15';

--4. Cho biết người thợ nào thực hiện công việc nhiều nhất.
SELECT TenTho
FROM THO 
WHERE MaTho = (
    SELECT TOP 1 MaTho
    FROM CHITIET_HD
    GROUP BY MaTho
    ORDER BY COUNT(MaCV) DESC
);

--5. Cho biết người thợ nào có tổng trị giá công việc được giao cao nhất.
SELECT TenTho
FROM THO
WHERE MaTho = (
    SELECT TOP 1 MaTho
    FROM CHITIET_HD
    GROUP BY MaTho
    ORDER BY SUM(TriGiaCV) DESC
);
